cmake_minimum_required(VERSION 3.16)
project(AdaptivePurePursuit VERSION 1.0 LANGUAGES CXX)
add_library(app_core cpp/app.cpp)
target_include_directories(app_core PUBLIC cpp)
target_compile_features(app_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(app_core PRIVATE /W4)
else()
  target_compile_options(app_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(app_demo cpp/main.cpp)
target_link_libraries(app_demo PRIVATE app_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(app_tests tests/test_core.cpp)
  target_link_libraries(app_tests PRIVATE app_core)
  add_test(NAME core COMMAND app_tests)
endif()
